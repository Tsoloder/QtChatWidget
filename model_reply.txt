这是一个使用 C++ 模板实现的快速排序算法。它支持任意可比较类型的容器（如 `std::vector`、`std::list` 等），并采用了标准的 Lomuto 分区方案或 Hoare 分区方案。这里提供一个基于 `std::vector` 的通用实现，使用递归方式，并允许自定义比较器（默认为小于号）。

```cpp
#ifndef QUICKSORT_H
#define QUICKSORT_H

#include <vector>
#include <functional>

namespace utils {

/**
 * @brief 快速排序算法实现
 * @tparam T 元素类型
 * @tparam Compare 比较函数对象类型，默认为 std::less<T>
 * @param arr 待排序的数组引用
 * @param compare 比较函数对象
 */
template <typename T, typename Compare = std::less<T>>
void quickSort(std::vector<T>& arr, Compare compare = Compare()) {
    if (arr.empty()) return;
    quickSortHelper(arr, 0, arr.size() - 1, compare);
}

/**
 * @brief 快速排序递归辅助函数
 */
template <typename T, typename Compare>
void quickSortHelper(std::vector<T>& arr, int low, int high, Compare compare) {
    if (low < high) {
        int pivotIndex = partition(arr, low, high, compare);
        quickSortHelper(arr, low, pivotIndex - 1, compare);
        quickSortHelper(arr, pivotIndex + 1, high, compare);
    }
}

/**
 * @brief 分区函数，使用 Lomuto 分区方案
 */
template <typename T, typename Compare>
int partition(std::vector<T>& arr, int low, int high, Compare compare) {
    T pivot = arr[high];
    int i = low - 1;

    for (int j = low; j < high; ++j) {
        if (compare(arr[j], pivot)) {
            ++i;
            std::swap(arr[i], arr[j]);
        }
    }
    std::swap(arr[i + 1], arr[high]);
    return i + 1;
}

} // namespace utils

#endif // QUICKSORT_H
```
