这是一个使用 C++ 计算斐波那契数列前 20 项的简单程序。我们将使用 `long long` 类型来防止整数溢出，并使用一个循环来迭代计算每一项。

```cpp
#include <iostream>
#include <vector>

int main() {
    const int n = 20;
    std::vector<long long> fib(n);

    // 初始化前两项
    if (n > 0) fib[0] = 0;
    if (n > 1) fib[1] = 1;

    // 计算后续项
    for (int i = 2; i < n; ++i) {
        fib[i] = fib[i - 1] + fib[i - 2];
    }

    // 输出结果
    std::cout << "Fibonacci sequence (first " << n << " terms):" << std::endl;
    for (int i = 0; i < n; ++i) {
        std::cout << "F(" << i << ") = " << fib[i] << std::endl;
    }

    return 0;
}
```
